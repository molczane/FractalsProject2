\section{Funkcja Weierstrassa}

% Slajd 4 - Funkcja Weierstrassa
\begin{frame}{Funkcja Weierstrassa}
    \begin{block}{Definicja}
        $$W(x) = \sum_{n=0}^{\infty} a^n \cos(b^n \pi x)$$
        gdzie $0 < a < 1$, $b > 1$, $ab > 1$
    \end{block}

    \textbf{Kluczowe właściwości:}
    \begin{itemize}
        \item Funkcja jest \textbf{ciągła}
        \item Funkcja jest \textbf{nigdzie nieróżniczkowalna}
        \item Wymiar fraktalny wykresu: $D = 2 + \frac{\log a}{\log b}$
    \end{itemize}

    \vspace{0.5em}
    \textbf{Przykład:} Dla $a=0.5$, $b=3$: $D \approx 1.369$
\end{frame}

\section{Wymiar Minkowskiego}

% Slajd 8 - Box-counting
\begin{frame}{Wymiar Minkowskiego (box-counting)}
    \begin{block}{Definicja}
        $$\dim_{box}(F) = \lim_{\epsilon \to 0} \frac{\log N(\epsilon)}{\log(1/\epsilon)}$$
    \end{block}

    \textbf{Algorytm:}
    \begin{enumerate}
        \item Pokryj obiekt siatką pudełek o rozmiarze $\epsilon$
        \item Policz niepuste pudełka $N(\epsilon)$
        \item Powtórz dla różnych $\epsilon$
        \item Wymiar = nachylenie prostej $\log N(\epsilon)$ vs $\log(1/\epsilon)$
    \end{enumerate}

    \vspace{0.5em}
    \textbf{Weryfikacja:}
    \begin{itemize}
        \item Linia: $D \approx 1.0$ \checkmark
        \item Kwadrat: $D \approx 2.0$ \checkmark
        \item Trójkąt Sierpińskiego: $D \approx 1.585$ \checkmark
    \end{itemize}
\end{frame}

\section{Fraktalne funkcje interpolacyjne}

% Slajd 9 - FIF
\begin{frame}{Fraktalne funkcje interpolacyjne (FIF)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Idea:}
            \begin{itemize}
                \item Interpolacja z fraktalną strukturą
                \item Kontrola ``szorstkości'' przez $d$
            \end{itemize}

            \vspace{1em}
            \textbf{Współczynnik $d$:}
            \begin{itemize}
                \item $d = 0$: interpolacja liniowa
                \item $|d| \to 1$: wymiar $\to 2$
            \end{itemize}

            \vspace{1em}
            \textbf{Zastosowania:}
            \begin{itemize}
                \item Modelowanie linii brzegowych
                \item Tekstury proceduralne
                \item Kompresja danych
            \end{itemize}
        \end{column}
