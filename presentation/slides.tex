\begin{frame}{Wprowadzenie: Spektrum interpolacji fraktalnej}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Główna idea:} FIF to pomost między interpolacją liniową a wypełnianiem przestrzeni.
            \vspace{0.3cm}

            \textbf{Parametr $d$ kontroluje szorstkość:}
            \begin{itemize}
                \item $d = 0$ $\rightarrow$ interpolacja liniowa ($D = 1$)
                \item $0 < |d| < 1$ $\rightarrow$ fraktalna struktura ($1 < D < 2$)
                \item $|d| \to 1$ $\rightarrow$ quasi-wypełnianie ($D \to 2$)
            \end{itemize}
            \vspace{0.3cm}

            \textbf{Związane koncepty:}
            \begin{itemize}
                \item Weierstrass $\approx$ prototyp struktury fraktalnej
                \item Peano/Hilbert $\approx$ przypadek graniczny
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/fif_basic.png}
        \end{column}
    \end{columns}
\end{frame}

%==============================================================================
% SLAJD 3: FIF - Definicja i konstrukcja IFS
%==============================================================================
\begin{frame}{FIF: Definicja i konstrukcja}
    \begin{block}{Fraktalna Funkcja Interpolacyjna}
        Funkcja ciągła $f: [x_0, x_n] \to \mathbb{R}$ przechodząca przez punkty $(x_i, y_i)$, będąca \textbf{atraktorem układu IFS}.
    \end{block}

    \vspace{0.3cm}
    \textbf{Transformacje afiniczne:}
    \[
    w_i(x, y) = \begin{pmatrix} a_i & 0 \\ e_i & d_i \end{pmatrix} \begin{pmatrix} x \\ y \end{pmatrix} + \begin{pmatrix} c_i \\ f_i \end{pmatrix}
    \]

    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Parametr $d_i \in (-1, 1)$:}
            \begin{itemize}
                \item Współczynnik skalowania pionowego
                \item Kontroluje ``szorstkość'' FIF
                \item Warunek zbieżności: $|d_i| < 1$
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Wymiar fraktalny:}
            \[
            \sum_{i=1}^{n} |d_i|^D = 1
            \]
            Dla równych $d$:
            \[
            D = \frac{\log n}{-\log |d|}
            \]
        \end{column}
    \end{columns}
\end{frame}

% SLAJD 5: Związek z funkcją Weierstrassa
%==============================================================================
\begin{frame}{Związek z funkcją Weierstrassa}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Funkcja Weierstrassa:}
            \[
            W(x) = \sum_{n=0}^{\infty} a^n \cos(b^n \pi x)
            \]

            \textbf{Wymiar fraktalny:}
            \[
            D = 2 + \frac{\log a}{\log b}
            \]

            \vspace{0.3cm}
            \textbf{Analogia z FIF:}
            \begin{itemize}
                \item $a$ w Weierstrassie $\leftrightarrow$ $d$ w FIF
                \item Oba kontrolują szorstkość
                \item Wymiar $\in (1, 2)$
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/weierstrass_interpolation.png}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{block}{Interpolacja z perturbacją Weierstrassa}
        Funkcja \texttt{weierstrass\_interpolation()} dodaje fraktalną strukturę do interpolacji liniowej.
    \end{block}
\end{frame}

\begin{frame}{Metoda box-counting: weryfikacja}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Algorytm:}
            \[
            D_{box} = \lim_{\varepsilon \to 0} \frac{\log N(\varepsilon)}{\log(1/\varepsilon)}
            \]

            \vspace{0.3cm}
            \textbf{Weryfikacja na znanych obiektach:}
            \begin{table}
                \small
                \begin{tabular}{lcc}
                    \toprule
                    Obiekt & $D_{theo}$ & $D_{num}$ \\
                    \midrule
                    Prosta & 1.0 & 0.99 \\
                    Sierpiński & 1.585 & 1.58 \\
                    Dywan & 1.893 & 1.87 \\
                    \bottomrule
                \end{tabular}
            \end{table}

            Błędy $< 5\%$ dla struktur samopodobnych.
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/sierpinski_box_counting.png}

%==============================================================================
% SLAJD 9: Analiza wymiaru FIF - teoria vs numeryka
%==============================================================================
\begin{frame}{Analiza wymiaru FIF: ograniczenia numeryczne}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/fif_dimension_vs_d.png}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Problem:} Dla wysokich $|d|$ wymiar numeryczny jest niedoszacowany.

            \vspace{0.3cm}
            \textbf{Przyczyny:}
            \begin{enumerate}
                \item \textbf{Gwałtowne oscylacje} -- amplituda rośnie wykładniczo
                \item \textbf{Efekt ``rozpadu''} -- punkty widoczne jako chmura
                \item \textbf{Ograniczony zakres $\varepsilon$} -- drobne struktury niewidoczne
            \end{enumerate}
        \end{column}
    \end{columns}

    \vspace{0.3cm}
    \begin{alertblock}{Wniosek}
        Box-counting jest skuteczny dla $|d| < 0.6$, ale wymaga ostrożnej interpretacji dla wyższych wartości.
    \end{alertblock}
\end{frame}

% SLAJD 10: Podsumowanie
%==============================================================================
\begin{frame}{Podsumowanie}
    \begin{enumerate}
        \item \textbf{FIF jako spektrum interpolacji:}
        \begin{itemize}
            \item $d = 0$ $\to$ liniowa ($D = 1$)
            \item $0 < |d| < 1$ $\to$ fraktalna ($1 < D < 2$)
            \item $|d| \to 1$ $\to$ quasi-wypełniająca ($D \to 2$)
        \end{itemize}

        \vspace{0.3cm}
        \item \textbf{Weierstrass} = prototyp struktury fraktalnej

        \vspace{0.3cm}
        \item \textbf{Krzywe Peano} = teoretyczna granica FIF

        \vspace{0.3cm}
        \item \textbf{Box-counting:} skuteczny dla umiarkowanych $d$, ograniczenia dla $|d| \to 1$
    \end{enumerate}

\end{frame}
