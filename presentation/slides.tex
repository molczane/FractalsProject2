\begin{frame}{Wprowadzenie: Po co interpolacja fraktalna?}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{Problem:} Jak modelować ``poszarpane'' wykresy?
            \begin{itemize}
                \item Notowania giełdowe
                \item Sygnały EKG
                \item Profile górskie, linie brzegowe
            \end{itemize}

            \vspace{0.3cm}
            \textbf{Klasyczna interpolacja} (wielomiany, splajny) daje \textbf{gładkie} krzywe -- nieodpowiednie dla danych fraktalnych.

            \vspace{0.3cm}
            \textbf{Rozwiązanie FIF:}
            \begin{itemize}
                \item Łączymy punkty \textbf{strukturą samopodobną}
                \item Wykres jest atraktorem układu IFS
                \item Kontrolujemy ``szorstkość'' parametrem
            \end{itemize}
        \end{column}

% SLAJD 3: Konstrukcja FIF - przekształcenia afiniczne
%==============================================================================
\begin{frame}{Konstrukcja FIF: Przekształcenia afiniczne}
    \textbf{Dane:} Punkty interpolacji $(x_0, y_0), (x_1, y_1), \ldots, (x_n, y_n)$.

    \vspace{0.2cm}
    \textbf{Idea:} Dzielimy odcinek $[x_0, x_n]$ na $n$ przedziałów i definiujemy przekształcenia $w_i$, które ``kopiują'' cały wykres w mniejsze przedziały.

    \vspace{0.2cm}
    \begin{block}{Przekształcenie afiniczne $w_i$}
        \[
        w_i(x, y) = \begin{pmatrix} a_i & 0 \\ e_i & d_i \end{pmatrix} \begin{pmatrix} x \\ y \end{pmatrix} + \begin{pmatrix} c_i \\ f_i \end{pmatrix}
        \]
        \begin{itemize}
            \item $a_i = \frac{x_i - x_{i-1}}{x_n - x_0}$ -- współczynnik ściskający dziedzinę
            \item $d_i \in (-1, 1)$ -- \textbf{współczynnik skalowania pionowego}
        \end{itemize}
    \end{block}

    \vspace{0.2cm}
    \begin{alertblock}{Rola parametru $d$}
        \begin{itemize}
            \item $d = 0$ $\Rightarrow$ interpolacja liniowa (brak struktury fraktalnej)
            \item $|d| \to 1$ $\Rightarrow$ coraz większa ``szorstkość'' wykresu
        \end{itemize}
    \end{alertblock}
\end{frame}

%==============================================================================
% SLAJD 4: Twierdzenie o istnieniu - Operator Read'a-Bajrakareviča
%==============================================================================
\begin{frame}{Istnienie rozwiązania: Operator Read'a-Bajrakareviča}
    \begin{theorem}[Istnienie i jednoznaczność FIF]
        Jeśli $|d_i| < 1$ dla wszystkich $i$, to operator $T$ zdefiniowany przez przekształcenia $w_i$ jest \textbf{kontrakcją} w przestrzeni funkcji ciągłych.
    \end{theorem}

    \vspace{0.3cm}
    \textbf{Konsekwencje (Twierdzenie Banacha):}
    \begin{enumerate}
        \item Istnieje \textbf{dokładnie jedna} funkcja ciągła $f$ spełniająca warunek samopodobieństwa.
        \item Funkcja $f$ przechodzi przez wszystkie punkty interpolacji.
        \item Wykres $f$ jest \textbf{atraktorem} układu IFS $\{w_1, \ldots, w_n\}$.
    \end{enumerate}

    \vspace{0.3cm}
    \begin{alertblock}{Warunek zbieżności}
        $|d_i| < 1$ dla wszystkich $i$ -- gwarantuje istnienie i jednoznaczność rozwiązania.
    \end{alertblock}
\end{frame}

%==============================================================================
% SLAJD 5: Wymiar fraktalny FIF - wzory
%==============================================================================
\begin{frame}{Wymiar fraktalny FIF: Wzory}
    Dla $n$ punktów równoodległych i jednakowych współczynników $|d|$:

    \vspace{0.3cm}
    \begin{block}{Przypadek ``gładki'' ($\sum |d_i| < 1$)}
        Wykres jest mało poszarpany:
        \[
        D = 1
        \]
    \end{block}

    \begin{block}{Przypadek fraktalny ($\sum |d_i| > 1$)}
        Wykres ma strukturę fraktalną:
        \[
        D = 1 + \frac{\log\left(\sum_{i=1}^{n} |d_i|\right)}{\log(n)}
        \]
    \end{block}

    \vspace{0.2cm}
    \textbf{Dla równych $|d|$:} $\sum |d_i| = n \cdot |d|$, więc:
    \[
    D = 1 + \frac{\log(n \cdot |d|)}{\log(n)} = 1 + 1 + \frac{\log |d|}{\log n} = 2 + \frac{\log |d|}{\log n}
    \]
\end{frame}

%==============================================================================
% SLAJD 6: Gładkość a wymiar - wykładnik Höldera
%==============================================================================
\begin{frame}{Gładkość a wymiar: Wykładnik Höldera}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Wykładnik Höldera $\alpha$:}

            Miara ``gładkości'' funkcji -- $f$ jest $\alpha$-hölderowska jeśli:
            \[
            |f(x) - f(y)| \leq C \cdot |x - y|^\alpha
            \]

            \vspace{0.3cm}
            \textbf{Związek z wymiarem:}
            \[
            \boxed{\alpha = 2 - D}
            \]

            \vspace{0.3cm}
            \textbf{Interpretacja:}
            \begin{itemize}
                \item $D \to 1$ $\Rightarrow$ $\alpha \to 1$ (gładka)
                \item $D \to 2$ $\Rightarrow$ $\alpha \to 0$ (``nerwowa'')
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/fif_spectrum.png}
        \end{column}
    \end{columns}
\end{frame}

% SLAJD 7: Związek z funkcją Weierstrassa
%==============================================================================
\begin{frame}{Związek z funkcją Weierstrassa}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Funkcja Weierstrassa:}
            \[
            W(x) = \sum_{n=0}^{\infty} a^n \cos(b^n \pi x)
            \]

            \textbf{Wymiar:} $D = 2 + \frac{\log a}{\log b}$

            \vspace{0.3cm}
            \textbf{Analogia z FIF:}
            \begin{itemize}
                \item Parametr $a$ $\leftrightarrow$ parametr $d$
                \item Oba kontrolują ``szorstkość''
                \item Wymiar $\in (1, 2)$
            \end{itemize}

            \vspace{0.2cm}
            \textbf{Weierstrass jako prototyp:}

            Struktura fraktalna między punktami FIF przypomina funkcję Weierstrassa.
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/weierstrass_interpolation.png}
        \end{column}
    \end{columns}
\end{frame}

% SLAJD 8: Krzywe Peano i FIF - wspólny framework IFS
%==============================================================================
\begin{frame}{Krzywe Peano jako funkcje interpolacji fraktalnej}
    \textbf{Kluczowe odkrycie (Barnsley, Massopust 1989):} Zarówno FIF jak i krzywe Peano są \textbf{atraktorami IFS}!

    \vspace{0.2cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Krzywa Hilberta jako interpolacja:}
            \begin{itemize}
                \item Odwzorowuje $t \in [0,1] \to (x(t), y(t)) \in [0,1]^2$
                \item Dla dowolnych punktów w kwadracie wyznacza \textbf{kolejność odwiedzin}
                \item Jest to \textbf{interpolacja parametryczna}
            \end{itemize}

            \vspace{0.2cm}
            \textbf{FIF przy $d \to 1$:}
            \begin{itemize}
                \item Wykres ``wypełnia'' coraz więcej przestrzeni
                \item $D \to 2$ (jak krzywa Peano)
                \item Zachowuje własność funkcji $y = f(x)$
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{assets/peano_fif_connection.png}
        \end{column}
    \end{columns}
\end{frame}

% SLAJD 10: Podsumowanie

\begin{frame}{Podsumowanie}
    \begin{enumerate}
        \item \textbf{FIF -- definicja:} Atraktor IFS, funkcja ciągła przechodząca przez punkty interpolacji z fraktalną strukturą między nimi.

        \vspace{0.2cm}
        \item \textbf{Parametr $d$:} Kontroluje szorstkość i wymiar fraktalny.

        \vspace{0.2cm}
        \item \textbf{Wymiar FIF:}
        \begin{itemize}
            \item $\sum |d_i| < 1 \Rightarrow D = 1$ (gładka)
            \item $\sum |d_i| > 1 \Rightarrow D = 1 + \frac{\log(\sum |d_i|)}{\log n}$
        \end{itemize}

        \vspace{0.2cm}
        \item \textbf{Gładkość:} $\alpha = 2 - D$ (wykładnik Höldera)

        \vspace{0.2cm}
        \item \textbf{Powiązania:} Weierstrass (prototyp), Peano (granica)
    \end{enumerate}
